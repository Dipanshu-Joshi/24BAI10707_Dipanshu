1.Insert one employee document.
Answer - db.employees.insertOne({ empId: 106, name: "Anita", age: 27, department: "IT", salary: 60000, city: "Pune", experience: 4, skills: ["Java","MongoDB"], active: true })
2.Insert multiple employee documents.
Answer - db.employees.insertMany([{ empId: 107, name: "Ravi", age: 29, department: "Finance", salary: 52000, city: "Hyderabad", experience: 5, skills: ["Excel"], active: true }, { empId: 108, name: "Meena", age: 24, department: "HR", salary: 48000, city: "Chennai", experience: 2, skills: ["Training"], active: true }])
3.Find one employee from IT department.
Answer - db.employees.findOne({ department: "IT" })
4.Display all employees.
Answer - db.employees.find()
5.Display only name and salary.
Answer - db.employees.find({}, { name: 1, salary: 1, _id: 0 })
6.Find employees with salary equal to 50000.
Answer - db.employees.find({ salary: 50000 })
7.Find employees with salary less than 60000.
Answer - db.employees.find({ salary: { $lt: 60000 } })
8.Find employees with salary greater than or equal to 60000.
Answer - db.employees.find({ salary: { $gte: 60000 } })
9.Find employees whose age is not equal to 30.
Answer - db.employees.find({ age: { $ne: 30 } })
10.Find employees from Chennai or Mumbai.
Answer - db.employees.find({ city: { $in: ["Chennai", "Mumbai"] } })
11.Find employees not from Chennai.
Answer - db.employees.find({ city: { $ne: "Chennai" } })
12.Find IT employees with salary > 50000.
Answer - db.employees.find({ department: "IT", salary: { $gt: 50000 } })
13.Find employees from Chennai OR Bangalore.
Answer - db.employees.find({ $or: [{ city: "Chennai" }, { city: "Bangalore" }] })
14.Find employees NOT earning above 60000.
Answer - db.employees.find({ salary: { $not: { $gt: 60000 } } })
15.Find employees neither in IT nor HR.
Answer - db.employees.find({ department: { $nin: ["IT", "HR"] } })
16.Find documents containing experience field.
Answer - db.employees.find({ experience: { $exists: true } })
17.Find documents where salary is Integer.
Answer - db.employees.find({ salary: { $type: "int" } })
18.Find employees having MongoDB skill.
Answer - db.employees.find({ skills: "MongoDB" })
19.Find employees having both MongoDB and AWS.
Answer - db.employees.find({ skills: { $all: ["MongoDB", "AWS"] } })
20.Find employees with exactly 3 skills.
Answer - db.employees.find({ skills: { $size: 3 } })
21.Find employees having Java skill using elemMatch.
Answer - db.employees.find({ skills: { $elemMatch: { $eq: "Java" } } })
22.Display employees sorted by salary descending.
Answer - db.employees.find().sort({ salary: -1 })
23.Display top 3 highest paid employees.
Answer - db.employees.find().sort({ salary: -1 }).limit(3)
24.Increase salary of employee 101 by 5000.
Answer - db.employees.updateOne({ empId: 101 }, { $inc: { salary: 5000 } })
25.Increase salary of all IT employees by 10%.
Answer - db.employees.updateMany({ department: "IT" }, { $mul: { salary: 1.1 } })
